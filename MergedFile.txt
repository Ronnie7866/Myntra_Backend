import asyncio

class CacheManager:
    def __init__(self, client: TelegramClient):
        self.client = client
        self.cleanup_interval = 3600  # 1 hour
        self._running = True

    async def verify_cached_file(self, file_id: int) -> bool:
        try:
            if not file_id or not str(file_id).isdigit():
                return False

            result = await self.client(
                GetMessagesRequest(channel=PRIVATE_CHAT_ID, id=[int(file_id)])
            )
            return bool(result and result.messages and result.messages[0].media)
        except Exception as e:
            log.error(f"Error verifying cached file {file_id}: {str(e)}")
            return False

    async def cleanup_stale_cache(self):
        try:
            log.info("Starting cache cleanup")
            keys_to_delete = set()

            # Get all keys that look like file IDs
            for key in db.scan_iter("*"):
                value = db.get(key)
                if not value:
                    continue

                if str(value).isdigit():
                    try:
                        if not await self.verify_cached_file(int(value)):
                            keys_to_delete.add(key)
                            keys_to_delete.add(f"mid_{value}")
                    except Exception as e:
                        log.error(f"Error checking key {key}: {str(e)}")
                        keys_to_delete.add(key)

            # Batch delete invalid keys
            for key in keys_to_delete:
                try:
                    db.delete(key)
                    log.info(f"Deleted stale cache key: {key}")
                except Exception as e:
                    log.error(f"Error deleting key {key}: {str(e)}")

            log.info(f"Cache cleanup completed. Removed {len(keys_to_delete)} stale entries")

        except Exception as e:
            log.error(f"Error in cache cleanup: {str(e)}")

    async def start_cleanup_task(self):
        while self._running:
            try:
                await self.cleanup_stale_cache()
                await asyncio.sleep(self.cleanup_interval)
            except asyncio.CancelledError:
                break
            except Exception as e:
                log.error(f"Error in cleanup task: {str(e)}")
                await asyncio.sleep(60)  # Wait a bit before retrying

    def stop(self):
        self._running = False

        elapsed_time = current_time - self.last_send_time

        if elapsed_time >= 5:

        else:

import asyncio
from typing import Dict
from logger_config import setup_logger
from collections import deque

log = setup_logger('download')


class DownloadManager:
    def __init__(self):
        self.active_downloads: Dict[int, asyncio.Task] = {}
        self.download_queue : deque = deque()
        self.max_active_downloads = 20
        
    async def add_download(self, user_id: int, download_coroutine):
        """Add a new download task for a user"""
        try:
            if user_id in self.active_downloads:
                if not self.active_downloads[user_id].done():
                    log.info(f"User {user_id} already has an active download")
                    return False
            if len(self.active_downloads) >= self.max_active_downloads:
                log.info(f"Download queue is full; adding user {user_id} to the queue")
                self.download_queue.append((user_id, download_coroutine))
                return "queued"

            # Create new download task
            task = asyncio.create_task(download_coroutine)
            self.active_downloads[user_id] = task
            log.info(f"Added new download task for user {user_id}")
            task.add_done_callback(lambda _:self._download_complete(user_id))
            return True
        except Exception as e:
            log.error(f"Error adding download for user {user_id}: {str(e)}")
            return False

    def _download_complete(self, user_id: int):
        """Handle cleanup after a download finishes"""
        log.info(f"Download completed for user {user_id}")
        self.active_downloads.pop(user_id, None)

        # Start the next download in the queue
        if self.download_queue:
            next_user, next_task = self.download_queue.popleft()
            log.info(f"Starting download for queued user {next_user}")
            asyncio.create_task(self.add_download(next_user, next_task))


    def get_active_download(self, user_id: int):
        """Get active download task for a user"""
        return self.active_downloads.get(user_id)
    
    def remove_download(self, user_id: int):
        """Remove a download task for a user"""
        if user_id in self.active_downloads:
            task = self.active_downloads.pop(user_id)
            if not task.done():
                task.cancel()
            log.info(f"Removed download task for user {user_id}")

class UploadSender:
    client: TelegramClient
    sender: MTProtoSender
    request: Union[SaveFilePartRequest, SaveBigFilePartRequest]
    part_count: int
    stride: int
    previous: Optional[asyncio.Task]
    loop: asyncio.AbstractEventLoop

    def __init__(
        self,
        client: TelegramClient,
        sender: MTProtoSender,
        file_id: int,
        part_count: int,
        big: bool,
        index: int,
        stride: int,
        loop: asyncio.AbstractEventLoop,
    ) -> None:
        self.client = client
        self.sender = sender
        self.part_count = part_count
        if big:
            self.request = SaveBigFilePartRequest(file_id, index, part_count, b"")
        else:
            self.request = SaveFilePartRequest(file_id, index, b"")
        self.stride = stride
        self.previous = None
        self.loop = loop

    async def next(self, data: bytes) -> None:
        if self.previous:
            await self.previous
        self.previous = self.loop.create_task(self._next(data))

    async def _next(self, data: bytes) -> None:
        self.request.bytes = data
        log.debug(
            f"Sending file part {self.request.file_part}/{self.part_count}"
            f" with {len(data)} bytes"
        )
        await self.client._call(self.sender, self.request)
        self.request.file_part += self.stride

    async def disconnect(self) -> None:
        if self.previous:
            await self.previous
        return await self.sender.disconnect()


class ParallelTransferrer:
    client: TelegramClient
    loop: asyncio.AbstractEventLoop
    dc_id: int
    senders: Optional[List[Union[UploadSender]]]
    auth_key: AuthKey
    upload_ticker: int

    def __init__(self, client: TelegramClient, dc_id: Optional[int] = None) -> None:
        self.client = client
        self.loop = self.client.loop
        self.dc_id = dc_id or self.client.session.dc_id
        self.auth_key = (
            None
            if dc_id and self.client.session.dc_id != dc_id
            else self.client.session.auth_key
        )
        self.senders = None
        self.upload_ticker = 0

    async def _cleanup(self) -> None:
        await asyncio.gather(*[sender.disconnect() for sender in self.senders])
        self.senders = None

    @staticmethod
    def _get_connection_count(
        file_size: int, max_count: int = 20, full_size: int = 100 * 1024 * 1024
    ) -> int:
        if file_size > full_size:
            return max_count
        return math.ceil((file_size / full_size) * max_count)

    async def _init_upload(
        self, connections: int, file_id: int, part_count: int, big: bool
    ) -> None:
        self.senders = [
            await self._create_upload_sender(file_id, part_count, big, 0, connections),
            *await asyncio.gather(
                *[
                    self._create_upload_sender(file_id, part_count, big, i, connections)
                    for i in range(1, connections)
                ]
            ),
        ]

    async def _create_upload_sender(
        self, file_id: int, part_count: int, big: bool, index: int, stride: int
    ) -> UploadSender:
        return UploadSender(
            self.client,
            await self._create_sender(),
            file_id,
            part_count,
            big,
            index,
            stride,
            loop=self.loop,
        )

    async def _create_sender(self) -> MTProtoSender:
        dc = await self.client._get_dc(self.dc_id)
        sender = MTProtoSender(self.auth_key, loggers=self.client._log)
        await sender.connect(
            self.client._connection(
                dc.ip_address,
                dc.port,
                dc.id,
                loggers=self.client._log,
                proxy=self.client._proxy,
            )
        )
        if not self.auth_key:
            log.debug(f"Exporting auth to DC {self.dc_id}")
            auth = await self.client(ExportAuthorizationRequest(self.dc_id))
            self.client._init_request.query = ImportAuthorizationRequest(
                id=auth.id, bytes=auth.bytes
            )
            req = InvokeWithLayerRequest(LAYER, self.client._init_request)
            await sender.send(req)
            self.auth_key = sender.auth_key
        return sender

    async def init_upload(
        self,
        file_id: int,
        file_size: int,
        part_size_kb: Optional[float] = None,
        connection_count: Optional[int] = None,
    ) -> Tuple[int, int, bool]:
        connection_count = connection_count or self._get_connection_count(file_size)
        part_size = (part_size_kb or utils.get_appropriated_part_size(file_size)) * 1024
        part_count = (file_size + part_size - 1) // part_size
        is_large = file_size > 10 * 1024 * 1024
        await self._init_upload(connection_count, file_id, part_count, is_large)
        return part_size, part_count, is_large

    async def upload(self, part: bytes) -> None:
        await self.senders[self.upload_ticker].next(part)
        self.upload_ticker = (self.upload_ticker + 1) % len(self.senders)

    async def finish_upload(self) -> None:
        await self._cleanup()


parallel_transfer_locks: DefaultDict[int, asyncio.Lock] = defaultdict(

def stream_file(file_to_stream: BinaryIO, chunk_size=1024):
    while True:
        data_read = file_to_stream.read(chunk_size)
        if not data_read:
            break
        yield data_read


async def _internal_transfer_to_telegram(
    client: TelegramClient,
    response: BinaryIO,
    progress_callback: callable,
    file_name: str = None,
) -> Tuple[TypeInputFile, int]:
    file_id = helpers.generate_random_long()
    file_size = os.path.getsize(response.name)

    hash_md5 = hashlib.md5()
    uploader = ParallelTransferrer(client)
    part_size, part_count, is_large = await uploader.init_upload(file_id, file_size)
    buffer = bytearray()
    for data in stream_file(response):
        if progress_callback:
            r = progress_callback(response.tell(), file_size)
            if inspect.isawaitable(r):
                await r
        if not is_large:
            hash_md5.update(data)
        if len(buffer) == 0 and len(data) == part_size:
            await uploader.upload(data)
            continue
        new_len = len(buffer) + len(data)
        if new_len >= part_size:
            cutoff = part_size - len(buffer)
            buffer.extend(data[:cutoff])
            await uploader.upload(bytes(buffer))
            buffer.clear()
            buffer.extend(data[cutoff:])
        else:
            buffer.extend(data)
    if len(buffer) > 0:
        await uploader.upload(bytes(buffer))
    await uploader.finish_upload()
    if is_large:
        return (
            InputFileBig(file_id, part_count, file_name if file_name else "upload"),
            file_size,
        )
    else:
        return (
            InputFile(
                file_id,
                part_count,
                file_name if file_name else "upload",
                hash_md5.hexdigest(),
            ),
            file_size,
        )


async def upload_file(
    client: TelegramClient,
    file: BinaryIO,
    progress_callback: callable = None,
    file_name: str = None,
) -> TypeInputFile:
    res = (
        await _internal_transfer_to_telegram(client, file, progress_callback, file_name)
    )[0]
    return res

import logging
import os
import sys
from logging.handlers import RotatingFileHandler

def setup_logger(name):
    log_dir = 'logs'
    os.makedirs(log_dir, exist_ok=True)

    logger = logging.getLogger(name)
    logger.setLevel(logging.DEBUG)

    # Clear any existing handlers if the logger has been initialized before
    if logger.handlers:
        for handler in logger.handlers[:]:
            logger.removeHandler(handler)

    # Console handler for stdout
    console_handler = logging.StreamHandler(sys.stdout)
    console_handler.setLevel(logging.DEBUG)

    # File handler with a specified filename
    file_handler = RotatingFileHandler(
        os.path.join(log_dir, f"{name}.log"),  # Specify the filename here
        maxBytes=10 * 1024 * 1024,  # 10 MB
        backupCount=5
    )
    file_handler.setLevel(logging.DEBUG)

    # Colored formatter for console output
    class ColoredFormatter(logging.Formatter):
        COLORS = {
            'DEBUG': '\033[94m',
            'INFO': '\033[92m',
            'WARNING': '\033[93m',
            'ERROR': '\033[91m',
            'CRITICAL': '\033[1;31m'
        }
        RESET = '\033[0m'

        def format(self, record):
            log_message = super().format(record)
            color = self.COLORS.get(record.levelname, '')
            return f"{color}{log_message}{self.RESET}"

    formatter = ColoredFormatter(
        '%(asctime)s | %(name)s | %(levelname)8s | %(message)s',
        datefmt='%Y-%m-%d %H:%M:%S'
    )

    # Plain formatter for file output
    file_formatter = logging.Formatter(
        '%(asctime)s | %(name)s | %(levelname)8s | %(message)s',
        datefmt='%Y-%m-%d %H:%M:%S'
    )

    # Assign formatters to handlers
    console_handler.setFormatter(formatter)
    file_handler.setFormatter(file_formatter)

    # Add handlers to logger
    logger.addHandler(console_handler)
    logger.addHandler(file_handler)

    return logger

async def handle_message(m: Message):
    try:
        # Check if user is admin
        is_admin = m.sender_id in ADMINS

        # Only check for active downloads for non-admin users
        if not is_admin:
            active_download = download_manager.get_active_download(m.sender_id)
            if active_download:

                )
                return

        url = get_urls_from_string(m.text)
        if not url:
            log.warning(f"No valid URL found in message from user {m.sender_id}")
            return await m.reply("Please enter a valid url.")

        log.info(f"Processing URL: {url} from user {m.sender_id}")
        hm = await m.reply("Processing your request, please wait...")

        shorturl = extract_code_from_url(url)
        if not shorturl:
            log.warning(f"Invalid Terabox link format for URL: {url}")
            return await hm.edit("Invalid Terabox link format.")

        fileid = db.get_key(shorturl)
        if fileid:
            log.info(f"Found cached file ID {fileid} for shorturl {shorturl}")
            uid = db.get_key(f"mid_{fileid}")
            check = await VideoSender.forward_file(
                file_id=fileid,
                message=m,
                client=bot,
                edit_message=hm,
                uid=uid
            )
            if check:
                if not is_admin:
                    db.increment_user_downloads(m.sender_id)
                return

        async def download_and_send():
            try:
                data = await get_data(url)
                if not data:
                    return await hm.edit(
                        "❌ Failed to get download link. Please try:\n"
                        "1. Check if the file still exists\n"
                        "2. Make sure the link is public\n"
                        "3. Try again later or try another link"
                    )

                    # Handle multiple videos
                if isinstance(data, list):
                    total_files = len(data)
                    await hm.edit(f"Found {total_files} files. Downloading and sending...")

                    for index, file_data in enumerate(data, 1):
                        if file_data['sizebytes'] < 1024 * 1024:
                            await hm.edit(f"Skipping file {index}: Invalid file size")
                            continue

                        sender = VideoSender(
                            client=bot,
                            data=file_data,
                            message=m,
                            edit_message=hm,
                            url=url,
                        )

                        await sender.send_video()
                        await asyncio.sleep(1)  # Small delay between files

                    # Existing single video handling
                else:
                    sender = VideoSender(
                        client=bot,
                        data=data,
                        message=m,
                        edit_message=hm,
                        url=url,
                    )
                    await sender.send_video()

            except Exception as e:
                log.error(f"Unexpected error in handle_message: {str(e)}")
                await hm.edit("An unexpected error occurred while processing the link.")

        # For admins, use create_task to allow multiple downloads
        # For non-admins, use the existing add_download method
        if is_admin:
            asyncio.create_task(download_and_send())
            return

        # Add download task to manager for non-admins
        success = await download_manager.add_download(m.sender_id, download_and_send())
        if success == "queued":
            await m.reply(
                "⚠️ The system is currently processing other downloads. Your request has been queued. "
                "Please wait, and you'll be notified when your download begins."
            )
        elif not success:
            await hm.edit(
                "⚠️ You already have an active download.\n"
                "Please wait for it to complete or use /cancel to stop it."
            )

    except Exception as e:
        log.error(f"Unexpected error in handle_message: {str(e)}")
        log.debug(traceback.format_exc())

            await safe_reply(event, "❌ Failed to check queue status.")
        except:
            # Fallback if even the error message can't be sent
            log.error("Could not send error message")


async def safe_reply(event, message, **kwargs):
    """
    Safely send a reply with built-in flood protection
    """
    max_retries = 3
    for attempt in range(max_retries):
        try:
            # Attempt to send the message
            return await event.reply(message, **kwargs)
        except FloodWaitError as flood:
            # Log the flood wait
            log.warning(f"Flood wait required. Seconds: {flood.seconds}")

            # Wait out the flood wait
            await asyncio.sleep(flood.seconds)
        except Exception as e:
            log.error(f"Error in safe_reply (attempt {attempt + 1}): {str(e)}")

            # If it's the last attempt, re-raise the exception
            if attempt == max_retries - 1:
                raise

            # Wait a bit before retrying
            await asyncio.sleep(2 ** attempt)

    # This should never be reached due to the max_retries logic
    raise Exception("Failed to send message after multiple attempts")


# Global error handler for Telethon
async def global_error_handler(error):
    """
    Global error handler to catch and log various errors
    """
    if isinstance(error, FloodWaitError):
        log.warning(f"Global Flood Wait: Waiting for {error.seconds} seconds")

        # Notify admins about the flood wait
        for admin in ADMINS:
            try:
                await bot.send_message(
                    admin,
                    f"⚠️ Bot experiencing Telegram API flood wait. "
                    f"Pausing for {error.seconds} seconds."
                )
            except Exception as notify_error:
                log.error(f"Could not notify admin: {notify_error}")

        # Wait out the flood
        await asyncio.sleep(error.seconds)
    else:
        # Log other types of errors
        log.error(f"Unhandled error: {error}")

    def __init__(self):
        self.active_downloads: Dict[int, asyncio.Task] = {}
        
    async def add_download(self, user_id: int, download_coroutine):
        """Add a new download task for a user"""
        try:
            if user_id in self.active_downloads:
                if not self.active_downloads[user_id].done():
                    log.info(f"User {user_id} already has an active download")
                    return False
                else:
                    # Clean up completed download
                    self.active_downloads.pop(user_id)
            
            # Create new download task
            task = asyncio.create_task(download_coroutine)
            self.active_downloads[user_id] = task
            log.info(f"Added new download task for user {user_id}")
            return True
            
        except Exception as e:
            log.error(f"Error adding download for user {user_id}: {str(e)}")
            return False
    
    def get_active_download(self, user_id: int):
        """Get active download task for a user"""
        return self.active_downloads.get(user_id)
    
    def remove_download(self, user_id: int):
        """Remove a download task for a user"""
        if user_id in self.active_downloads:
            task = self.active_downloads.pop(user_id)
            if not task.done():
                task.cancel()
            log.info(f"Removed download task for user {user_id}")

class UploadSender:
    client: TelegramClient
    sender: MTProtoSender
    request: Union[SaveFilePartRequest, SaveBigFilePartRequest]
    part_count: int
    stride: int
    previous: Optional[asyncio.Task]
    loop: asyncio.AbstractEventLoop

    def __init__(
        self,
        client: TelegramClient,
        sender: MTProtoSender,
        file_id: int,
        part_count: int,
        big: bool,
        index: int,
        stride: int,
        loop: asyncio.AbstractEventLoop,
    ) -> None:
        self.client = client
        self.sender = sender
        self.part_count = part_count
        if big:
            self.request = SaveBigFilePartRequest(file_id, index, part_count, b"")
        else:
            self.request = SaveFilePartRequest(file_id, index, b"")
        self.stride = stride
        self.previous = None
        self.loop = loop

    async def next(self, data: bytes) -> None:
        if self.previous:
            await self.previous
        self.previous = self.loop.create_task(self._next(data))

    async def _next(self, data: bytes) -> None:
        self.request.bytes = data
        log.debug(
            f"Sending file part {self.request.file_part}/{self.part_count}"
            f" with {len(data)} bytes"
        )
        await self.client._call(self.sender, self.request)
        self.request.file_part += self.stride

    async def disconnect(self) -> None:
        if self.previous:
            await self.previous
        return await self.sender.disconnect()


class ParallelTransferrer:
    client: TelegramClient
    loop: asyncio.AbstractEventLoop
    dc_id: int
    senders: Optional[List[Union[UploadSender]]]
    auth_key: AuthKey
    upload_ticker: int

    def __init__(self, client: TelegramClient, dc_id: Optional[int] = None) -> None:
        self.client = client
        self.loop = self.client.loop
        self.dc_id = dc_id or self.client.session.dc_id
        self.auth_key = (
            None
            if dc_id and self.client.session.dc_id != dc_id
            else self.client.session.auth_key
        )
        self.senders = None
        self.upload_ticker = 0

    async def _cleanup(self) -> None:
        await asyncio.gather(*[sender.disconnect() for sender in self.senders])
        self.senders = None

    @staticmethod
    def _get_connection_count(
        file_size: int, max_count: int = 20, full_size: int = 100 * 1024 * 1024
    ) -> int:
        if file_size > full_size:
            return max_count
        return math.ceil((file_size / full_size) * max_count)

    async def _init_upload(
        self, connections: int, file_id: int, part_count: int, big: bool
    ) -> None:
        self.senders = [
            await self._create_upload_sender(file_id, part_count, big, 0, connections),
            *await asyncio.gather(
                *[
                    self._create_upload_sender(file_id, part_count, big, i, connections)
                    for i in range(1, connections)
                ]
            ),
        ]

    async def _create_upload_sender(
        self, file_id: int, part_count: int, big: bool, index: int, stride: int
    ) -> UploadSender:
        return UploadSender(
            self.client,
            await self._create_sender(),
            file_id,
            part_count,
            big,
            index,
            stride,
            loop=self.loop,
        )

    async def _create_sender(self) -> MTProtoSender:
        dc = await self.client._get_dc(self.dc_id)
        sender = MTProtoSender(self.auth_key, loggers=self.client._log)
        await sender.connect(
            self.client._connection(
                dc.ip_address,
                dc.port,
                dc.id,
                loggers=self.client._log,
                proxy=self.client._proxy,
            )
        )
        if not self.auth_key:
            log.debug(f"Exporting auth to DC {self.dc_id}")
            auth = await self.client(ExportAuthorizationRequest(self.dc_id))
            self.client._init_request.query = ImportAuthorizationRequest(
                id=auth.id, bytes=auth.bytes
            )
            req = InvokeWithLayerRequest(LAYER, self.client._init_request)
            await sender.send(req)
            self.auth_key = sender.auth_key
        return sender

    async def init_upload(
        self,
        file_id: int,
        file_size: int,
        part_size_kb: Optional[float] = None,
        connection_count: Optional[int] = None,
    ) -> Tuple[int, int, bool]:
        connection_count = connection_count or self._get_connection_count(file_size)
        part_size = (part_size_kb or utils.get_appropriated_part_size(file_size)) * 1024
        part_count = (file_size + part_size - 1) // part_size
        is_large = file_size > 10 * 1024 * 1024
        await self._init_upload(connection_count, file_id, part_count, is_large)
        return part_size, part_count, is_large

    async def upload(self, part: bytes) -> None:
        await self.senders[self.upload_ticker].next(part)
        self.upload_ticker = (self.upload_ticker + 1) % len(self.senders)

    async def finish_upload(self) -> None:
        await self._cleanup()

def stream_file(file_to_stream: BinaryIO, chunk_size=1024):
    while True:
        data_read = file_to_stream.read(chunk_size)
        if not data_read:
            break
        yield data_read


async def _internal_transfer_to_telegram(
    client: TelegramClient,
    response: BinaryIO,
    progress_callback: callable,
    file_name: str = None,
) -> Tuple[TypeInputFile, int]:
    file_id = helpers.generate_random_long()
    file_size = os.path.getsize(response.name)

    hash_md5 = hashlib.md5()
    uploader = ParallelTransferrer(client)
    part_size, part_count, is_large = await uploader.init_upload(file_id, file_size)
    buffer = bytearray()
    for data in stream_file(response):
        if progress_callback:
            r = progress_callback(response.tell(), file_size)
            if inspect.isawaitable(r):
                await r
        if not is_large:
            hash_md5.update(data)
        if len(buffer) == 0 and len(data) == part_size:
            await uploader.upload(data)
            continue
        new_len = len(buffer) + len(data)
        if new_len >= part_size:
            cutoff = part_size - len(buffer)
            buffer.extend(data[:cutoff])
            await uploader.upload(bytes(buffer))
            buffer.clear()
            buffer.extend(data[cutoff:])
        else:
            buffer.extend(data)
    if len(buffer) > 0:
        await uploader.upload(bytes(buffer))
    await uploader.finish_upload()
    if is_large:
        return (
            InputFileBig(file_id, part_count, file_name if file_name else "upload"),
            file_size,
        )
    else:
        return (
            InputFile(
                file_id,
                part_count,
                file_name if file_name else "upload",
                hash_md5.hexdigest(),
            ),
            file_size,
        )


async def upload_file(
    client: TelegramClient,
    file: BinaryIO,
    progress_callback: callable = None,
    file_name: str = None,
) -> TypeInputFile:
    res = (
        await _internal_transfer_to_telegram(client, file, progress_callback, file_name)
    )[0]
    return res

import sys
from logging.handlers import RotatingFileHandler

def setup_logger(name):
    log_dir = 'logs'
    os.makedirs(log_dir, exist_ok=True)

    logger = logging.getLogger(name)
    logger.setLevel(logging.DEBUG)

    # Clear any existing handlers if the logger has been initialized before
    if logger.handlers:
        for handler in logger.handlers[:]:
            logger.removeHandler(handler)

    # Console handler for stdout
    console_handler = logging.StreamHandler(sys.stdout)
    console_handler.setLevel(logging.DEBUG)

    # File handler with a specified filename
    file_handler = RotatingFileHandler(
        os.path.join(log_dir, f"{name}.log"),  # Specify the filename here
        maxBytes=10 * 1024 * 1024,  # 10 MB
        backupCount=5
    )
    file_handler.setLevel(logging.DEBUG)

    # Colored formatter for console output
    class ColoredFormatter(logging.Formatter):
        COLORS = {
            'DEBUG': '\033[94m',
            'INFO': '\033[92m',
            'WARNING': '\033[93m',
            'ERROR': '\033[91m',
            'CRITICAL': '\033[1;31m'
        }
        RESET = '\033[0m'

        def format(self, record):
            log_message = super().format(record)
            color = self.COLORS.get(record.levelname, '')
            return f"{color}{log_message}{self.RESET}"

    formatter = ColoredFormatter(
        '%(asctime)s | %(name)s | %(levelname)8s | %(message)s',
        datefmt='%Y-%m-%d %H:%M:%S'
    )

    # Plain formatter for file output
    file_formatter = logging.Formatter(
        '%(asctime)s | %(name)s | %(levelname)8s | %(message)s',
        datefmt='%Y-%m-%d %H:%M:%S'
    )

    # Assign formatters to handlers
    console_handler.setFormatter(formatter)
    file_handler.setFormatter(file_formatter)

    # Add handlers to logger
    logger.addHandler(console_handler)
    logger.addHandler(file_handler)

    return logger

import redis

class RedisCache:
    def __init__(self, host='localhost', port=6379):
        self.redis_client = redis.Redis(host=host, port=port)

    def store_video_metadata(self, video_id: str, metadata: Dict[str, Any]):
        """
        Store complete video metadata in Redis
        """
        self.redis_client.hmset(f"video:{video_id}", {
            'file_path': metadata.get('file_path', ''),
            'file_size': metadata.get('file_size', 0),
            'download_url': metadata.get('download_url', ''),
            'timestamp': metadata.get('timestamp', '')
        })
        # Set expiration to 7 days
        self.redis_client.expire(f"video:{video_id}", 7 * 24 * 60 * 60)

    def get_video_metadata(self, video_id: str) -> Dict[str, Any]:
        """
        Retrieve video metadata from Redis
        """
        metadata = self.redis_client.hgetall(f"video:{video_id}")
        if metadata:
            return {
                k.decode(): v.decode()
                for k, v in metadata.items()
            }
        return {}

    def check_video_exists(self, video_id: str) -> bool:
        """
        Check if video metadata exists in cache
        """
        return self.redis_client.exists(f"video:{video_id}")

import logging

class Redis(r):
    def __init__(
            self,
            host: str = None,
            port: int = None,
            password: str = None,
            logger=None,
            encoding: str = "utf-8",
            decode_responses=True,
            **kwargs,
    ):
        try:
            log.info("Initializing Redis connection")

            # Enhanced host and port parsing
            if ":" in host:
                data = host.split(":")
                host = data[0]
                port = int(data[1])

            if host.startswith("http"):
                log.error("REDIS_URI should not start with http!")
                sys.exit(1)

            if not host or not port:
                log.error("Host or Port Number not found")
                sys.exit(1)

            # Prepare connection kwargs
            kwargs["host"] = host
            if password and len(password) > 1:
                kwargs["password"] = password
            kwargs["port"] = port
            kwargs["encoding"] = encoding
            kwargs["decode_responses"] = decode_responses

            # Attempt Redis connection
            try:
                super().__init__(**kwargs)
                log.info(f"Connected to Redis at {host}:{port}")
            except Exception as e:
                log.critical(f"Failed to connect to Redis: {str(e)}")
                sys.exit(1)

            # Initialize cache
            self.logger = logger or log
            self._cache = {}

            # Start cache refresh thread
            threading.Thread(target=self.re_cache, daemon=True).start()

        except Exception as e:
            log.error(f"Unexpected error in Redis initialization: {str(e)}")
            log.debug(traceback.format_exc())
            sys.exit(1)

    def re_cache(self):
        """
        Refresh the local cache with Redis keys
        """
        try:
            log.debug("Starting cache refresh")
            keys = self.keys()
            self._cache = {key: self.get(key) for key in keys}
            log.info(f"Cached {len(self._cache)} keys")
        except Exception as e:
            log.error(f"Error during cache refresh: {str(e)}")
            log.debug(traceback.format_exc())

    def get_key(self, key: Any):
        """
        Retrieve a key from cache or Redis
        """
        try:
            if key in self._cache:
                log.debug(f"Retrieved key {key} from cache")
                return self._cache[key]

            data = self.get(key)
            if data is not None:
                log.debug(f"Retrieved key {key} from Redis")
                self._cache[key] = data

            return data
        except Exception as e:
            log.error(f"Error retrieving key {key}: {str(e)}")
            log.debug(traceback.format_exc())
            return None

    def del_key(self, key: Any):
        """
        Delete a key from cache and Redis
        """
        try:
            log.info(f"Deleting key {key}")
            if key in self._cache:
                del self._cache[key]
            return self.delete(key)
        except Exception as e:
            log.error(f"Error deleting key {key}: {str(e)}")
            log.debug(traceback.format_exc())
            return False

    def set_key(self, key: Any = None, value: Any = None, **kwargs):
        """
        Set a key in cache and Redis
        """
        try:
            log.info(f"Setting key {key}")
            self._cache[key] = value
            return self.set(key, value, **kwargs)
        except Exception as e:
            log.error(f"Error setting key {key}: {str(e)}")
            log.debug(traceback.format_exc())
            return False

    def incr(self, key, amount=1):
        """
        Increment a key's value
        """
        try:
            log.debug(f"Incrementing key {key} by {amount}")
            result = super().incr(key, amount)
            self._cache[key] = result
            return result
        except Exception as e:
            log.error(f"Error incrementing key {key}: {str(e)}")
            log.debug(traceback.format_exc())
            return 0

    def increment_user_downloads(self, user_id: int) -> int:
        """Increment download count for user and return total downloads in last 24h"""
        key = f"user_downloads:{user_id}"
        pipe = self.pipeline()
        pipe.incr(key)
        pipe.expire(key, 86400)  # 24 hours in seconds
        result = pipe.execute()
        return result[0]

    def get_user_downloads(self, user_id: int) -> int:
        """Get total downloads for user in last 24h"""
        key = f"user_downloads:{user_id}"
        count = self.get(key)
        return int(count) if count else 0

def find_between(data: str, first: str, last: str) -> str | None:
    """
    Searches for the first occurrence of the `first` string in `data`,
    and returns the text between the two strings.
    """
    try:
        start = data.index(first) + len(first)
        end = data.index(last, start)
        return data[start:end]
    except ValueError:
        log.warning(f"Could not find text between '{first}' and '{last}'")
        return None

        response = session.get(url, headers=headersList)
        response = session.get(response.url, headers=headersList)

        text = response.text
        default_thumbnail = find_between(text, 'og:image" content="', '"')
        logid = find_between(text, "dp-logid=", "&")

        response = session.get(reqUrl, headers=headersList)
        r_j = response.json()

            return False

        if r_j["errno"]:
            return False

        if "list" not in r_j and not r_j["list"]:

def get_formatted_size(size_bytes: int) -> str:
    """
    Returns a human-readable file size.
    """
    try:
        if size_bytes >= 1024 * 1024:
            size = size_bytes / (1024 * 1024)
            unit = "MB"
        elif size_bytes >= 1024:
            size = size_bytes / 1024
            unit = "KB"
        else:
            size = size_bytes
            unit = "b"

        formatted_size = f"{size:.2f} {unit}"
        log.debug(f"Converted {size_bytes} bytes to {formatted_size}")
        return formatted_size
    except Exception as e:
        log.error(f"Error in get_formatted_size: {str(e)}")
        log.debug(traceback.format_exc())
        return "0 b"


def convert_seconds(seconds: int) -> str:
    """
    Convert seconds into a human-readable format.
    """
    try:
        seconds = int(seconds)
        hours = seconds // 3600
        remaining_seconds = seconds % 3600
        minutes = remaining_seconds // 60
        remaining_seconds_final = remaining_seconds % 60

        if hours > 0:
            result = f"{hours}h:{minutes}m:{remaining_seconds_final}s"
        elif minutes > 0:
            result = f"{minutes}m:{remaining_seconds_final}s"
        else:
            result = f"{remaining_seconds_final}s"

        log.debug(f"Converted {seconds} seconds to {result}")
        return result
    except Exception as e:
        log.error(f"Error in convert_seconds: {str(e)}")
        log.debug(traceback.format_exc())
        return "0s"

def download_image_to_bytesio(url: str, filename: str) -> BytesIO | None:
    """
    Downloads an image from a URL and returns it as a BytesIO object.
    """
    try:
        log.info(f"Downloading image from {url}")
        response = requests.get(url)
        content = BytesIO()
        content.name = filename

        if response.status_code == 200:
            for chunk in response.iter_content(chunk_size=1024):
                content.write(chunk)
            log.info(f"Successfully downloaded image to {filename}")
        else:
            log.warning(f"Failed to download image. Status code: {response.status_code}")
            return None

        content.seek(0)
        return content
    except Exception as e:
        log.error(f"Error downloading image: {str(e)}")
        log.debug(traceback.format_exc())
        return None

async def check_force_sub(client, user_id, channel_username, group_username):
    """
    Check if a user has subscribed to both a specific channel and group.
    """
    try:
        log.info(f"Checking subscription for user {user_id} on channel {channel_username} and group {group_username}")

        # Check channel subscription
        try:
            channel = await client.get_entity(channel_username)
            channel_participant = await client(GetParticipantRequest(
                channel=channel,
                participant=user_id
            ))

            if isinstance(channel_participant.participant, (ChannelParticipantLeft, ChannelParticipantBanned)):
                log.warning(f"User {user_id} is left or banned from channel")
                return False
        except Exception as channel_error:
            log.error(f"Error checking channel subscription: {str(channel_error)}")
            return False

        # Check group subscription
        try:
            group = await client.get_entity(group_username)
            group_participant = await client(GetParticipantRequest(
                channel=group,
                participant=user_id
            ))

            if isinstance(group_participant.participant, (ChannelParticipantLeft, ChannelParticipantBanned)):
                log.warning(f"User {user_id} is left or banned from group")
                return False
        except Exception as group_error:
            log.error(f"Error checking group subscription: {str(group_error)}")
            return False

        log.info(f"User {user_id} is a valid participant in both channel and group")
        return True

    except Exception as e:
        log.error(f"Unexpected error in check_force_sub: {str(e)}")
        log.debug(traceback.format_exc())
        return False

from redis_db import db
import json
import time
from logger_config import setup_logger

log = setup_logger('video_coordinator')

class VideoCoordinator:
    def __init__(self, redis_client):
        self.redis = redis_client
        self.processing_key = "processing_videos"
        self.cache_key = "video_cache"
        self.lock_timeout = 300  # 5 minutes
        self.cache_timeout = 3600  # 1 hour

    async def is_video_processing(self, video_url: str) -> bool:
        """Check if video is currently being processed"""
        return bool(self.redis.get(f"{self.processing_key}:{video_url}"))

    async def set_video_processing(self, video_url: str) -> None:
        """Mark video as being processed"""
        self.redis.set(f"{self.processing_key}:{video_url}", "1", ex=self.lock_timeout)

    async def remove_video_processing(self, video_url: str) -> None:
        """Remove processing mark for video"""
        self.redis.delete(f"{self.processing_key}:{video_url}")

    async def cache_video(self, video_url: str, file_id: str) -> None:
        """Cache video file_id"""
        self.redis.set(f"{self.cache_key}:{video_url}", file_id, ex=self.cache_timeout)

    async def get_cached_video(self, video_url: str) -> str:
        """Get cached video file_id"""
        return self.redis.get(f"{self.cache_key}:{video_url}")

    async def remove_from_cache(self, video_url: str) -> None:
        """Remove video from cache"""
        self.redis.delete(f"{self.cache_key}:{video_url}")
